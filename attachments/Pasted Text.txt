    // Audio Context Setup for Synthesized Sounds
    const AudioContext = window.AudioContext || window.webkitAudioContext;
    let audioCtx = null;

    function initAudio() {
        if (!audioCtx) audioCtx = new AudioContext();
        if (audioCtx.state === 'suspended') audioCtx.resume();
    }

    function playBrakeSound() {
        if (!audioCtx) return;
        const bufferSize = audioCtx.sampleRate * 0.2;
        const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
        const data = buffer.getChannelData(0);
        for (let i = 0; i < bufferSize; i++) data[i] = Math.random() * 2 - 1;

        const noise = audioCtx.createBufferSource();
        noise.buffer = buffer;
        const filter = audioCtx.createBiquadFilter();
        filter.type = 'bandpass';
        filter.frequency.value = 2800;
        filter.Q.value = 6;

        const gain = audioCtx.createGain();
        gain.gain.setValueAtTime(0.12, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.2);

        noise.connect(filter);
        filter.connect(gain);
        gain.connect(audioCtx.destination);
        noise.start();
    }

    function playChimeSound() {
        if (!audioCtx) return;
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();

        osc.type = 'sine';
        osc.frequency.setValueAtTime(523.25, audioCtx.currentTime); // C5
        osc.frequency.setValueAtTime(659.25, audioCtx.currentTime + 0.12); // E5

        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.4);

        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + 0.4);
    }

    // Controls
    throttleEl.addEventListener('input', (e) => {
        initAudio();
        if (doorsOpen) {
            showLog("⚠️ Close doors before accelerating!");
            throttleEl.value = 0;
            return;
        }
        if (isEmergencyBraking || routeFinished) return;
        const level = parseInt(e.target.value);

        // Detect harsh throttle changes (> 3 levels at once)
        if (Math.abs(level - prevThrottle) > 3) {
            accelDeductions += 5;
            showLog("⚠️ Jerky acceleration/deceleration! (-Comfort)");
        }
        prevThrottle = level;

        targetSpeed = level * 6 * direction;
    });

            showLog("⚠️ Heavy braking! (-Comfort)");
        }
        throttleEl.value = Math.max(0, currentThrot - 2);
        prevThrottle = parseInt(throttleEl.value);
        targetSpeed = (prevThrottle * 6) * direction;

    // Formatting Stopwatch Time
    function formatTime(ms) {
        const totalSec = ms / 1000;
        const min = Math.floor(totalSec / 60);
        const sec = Math.floor(totalSec % 60);
        const tenths = Math.floor((ms % 1000) / 100);
        return `${min.toString().padStart(2, '0')}:${sec.toString().padStart(2, '0')}.${tenths}`;
    }

    // Daytime Side-View Rendering
    const groundY = 240;

    function drawDaytimeBackground(cameraX) {
        // Bright Daytime Sky
        ctx.fillStyle = '#38bdf8';
        ctx.fillRect(0, 0, canvas.width, canvas.height);

        // Sun
        ctx.fillStyle = '#fef08a';
        ctx.beginPath();
        ctx.arc(850, 60, 36, 0, Math.PI * 2);
        ctx.fill();

        // Fluffy Clouds (Parallax 0.05)
        ctx.fillStyle = '#ffffff';
        [200, 600, 1100, 1600, 2200, 2800, 3500, 4200, 5000].forEach((cx, idx) => {
            const screenX = cx - (cameraX * 0.05);
            if (screenX > -150 && screenX < canvas.width + 150) {
                const cy = 50 + (idx % 3) * 20;
                ctx.beginPath();
                ctx.arc(screenX, cy, 22, 0, Math.PI * 2);
                ctx.arc(screenX + 25, cy - 10, 28, 0, Math.PI * 2);
                ctx.arc(screenX + 55, cy, 22, 0, Math.PI * 2);
                ctx.fill();
            }
        });

        // Distant Green Hills (Parallax 0.15)
        ctx.fillStyle = '#86efac';
        ctx.beginPath();
        ctx.moveTo(0, groundY);
        for (let x = -200; x < canvas.width + 200; x += 100) {
            const worldX = x + (cameraX * 0.15);
            const height = 50 + Math.sin(worldX * 0.003) * 30 + Math.cos(worldX * 0.007) * 20;
            ctx.lineTo(x, groundY - height);
        }
        ctx.lineTo(canvas.width, groundY);
        ctx.fill();

        // Midground Trees (Parallax 0.4)
        for (let x = 0; x < 11000; x += 220) {
            const screenX = x - (cameraX * 0.4);
            if (screenX > -100 && screenX < canvas.width + 100) {
                // Tree Trunk
                ctx.fillStyle = '#b45309';
                ctx.fillRect(screenX - 4, groundY - 60, 8, 60);
                // Foliage
                ctx.fillStyle = '#15803d';
                ctx.beginPath();
                ctx.arc(screenX, groundY - 80, 26, 0, Math.PI * 2);
                ctx.fill();
                ctx.fillStyle = '#22c55e';
                ctx.beginPath();
                ctx.arc(screenX - 8, groundY - 88, 18, 0, Math.PI * 2);
                ctx.fill();
            }
        }
    }

    function drawTrackAndScenery(cameraX) {
        // Ground Layer
        ctx.fillStyle = '#15803d';
        ctx.fillRect(0, groundY, canvas.width, canvas.height - groundY);

        // Track Bed Gravel
        ctx.fillStyle = '#64748b';
        ctx.fillRect(0, groundY - 8, canvas.width, 8);

        // Ties & Overhead Catenary Poles
        const startX = Math.floor((cameraX - 100) / 20) * 20;
        for (let x = startX; x < cameraX + canvas.width + 100; x += 20) {
            const screenX = x - cameraX;
            // Wooden Ties
            ctx.fillStyle = '#78350f';
            ctx.fillRect(screenX - 4, groundY - 12, 8, 4);

            // Overhead Poles
            if (Math.abs(x) % 240 === 0) {
                ctx.fillStyle = '#94a3b8';
                ctx.fillRect(screenX - 3, groundY - 120, 6, 110);
                ctx.fillRect(screenX - 3, groundY - 120, 40, 5);
            }
        }

        // Catenary Overhead Wire
        ctx.strokeStyle = '#475569';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(0, groundY - 115);
        ctx.lineTo(canvas.width, groundY - 115);
        ctx.stroke();

        // Steel Rail
        ctx.fillStyle = '#cbd5e1';
        ctx.fillRect(0, groundY - 16, canvas.width, 4);
    }

                // LEGO Yellow Safety Line

                // Canopy / Roof

    function drawTrainSideView(cameraX) {
        const carSpacing = 190;
        const trainX = trainPos - cameraX;

        drawLegoCarriageSide(trainX - carSpacing * 2, '#ffcf00', false, true);
        drawLegoCarriageSide(trainX - carSpacing, '#0055bf', false, false);

    }

    function triggerTerminalFinish() {
        timerRunning = false;
        routeFinished = true;
        throttleEl.value = 0;
        targetSpeed = 0;

        // Calculate Intermediate Stations Skipped
        const skippedCount = stations.filter(s => !s.isTerminal && !s.visited).length;
        const skippedPenalty = skippedCount * 150;

        // Calculate Comfort Level Score (Base 100)
        const accelPen = accelDeductions;
        const revPen = reversalCount * 5;
        const midStopPen = midTrackStopCount * 10;
        const comfortScore = Math.max(0, 100 - accelPen - revPen - midStopPen);

        // Calculate Combined Total Score
        // Base Points: Station Points (100 pts each) + Passengers Delivered (20 pts each) + Comfort Score
        const pointsBase = (stationPoints * 100) + (passengersDelivered * 20);
        const timePenalty = Math.floor(elapsedTime / 1000); // 1 pt deduction per second
        const totalScore = Math.max(0, pointsBase + comfortScore - timePenalty - skippedPenalty);

        document.getElementById('res-time').innerText = formatTime(elapsedTime);
        document.getElementById('res-points').innerText = `${stationPoints} / 8 (+${stationPoints * 100} pts)`;
        document.getElementById('res-skipped-penalty').innerText = `-${skippedPenalty} pts (${skippedCount} skipped)`;
        document.getElementById('res-passengers').innerText = `${passengersDelivered} (+${passengersDelivered * 20} pts)`;
        document.getElementById('res-accel-penalty').innerText = `-${accelPen} pts`;
        document.getElementById('res-reversal-penalty').innerText = `-${revPen} pts (${reversalCount}x)`;
        document.getElementById('res-midstop-penalty').innerText = `-${midStopPen} pts (${midTrackStopCount}x)`;
        document.getElementById('res-comfort').innerText = `${comfortScore} / 100`;
        document.getElementById('res-total-score').innerText = `${totalScore} PTS`;

        document.getElementById('summary-modal').style.display = 'flex';
    }

    // Main Loop
    function update(timestamp) {
        if (!startTime) startTime = timestamp;

        if (timerRunning) {
            elapsedTime = timestamp - startTime;
            stopwatchValEl.innerText = formatTime(elapsedTime);
        }

        // Dynamics & Acceleration
        if (isEmergencyBraking) {
            trainSpeed += (0 - trainSpeed) * 0.15;
        } else {
            const accel = targetSpeed > trainSpeed ? 0.05 : 0.08;
            trainSpeed += (targetSpeed - trainSpeed) * accel;
        }

        if (Math.abs(trainSpeed) < 0.01) trainSpeed = 0;

        // Track Mid-Track Stops (stops away from any station platform)
        const atStation = stations.some(s => Math.abs(s.pos - trainPos) < 100);
        if (trainSpeed === 0 && !wasStopped) {
            wasStopped = true;
            if (!atStation && trainPos > 150 && !routeFinished) {
                midTrackStopCount++;
                showLog("⚠️ Stopped in middle of track! (-10 Comfort)");
            }
        } else if (Math.abs(trainSpeed) > 0.5) {
            wasStopped = false;
        }

        // Position Updates
        trainPos += trainSpeed * 0.4;
        wheelRotation += (trainSpeed * 0.05);

        // Track limits
        if (trainPos < 0) trainPos = 0;

        // Camera Tracking Centered on Engine
        const cameraX = trainPos - 400;

        speedValEl.innerText = Math.round(Math.abs(trainSpeed));
        onboardValEl.innerText = passengersOnBoard;
        passengersValEl.innerText = passengersDelivered;
        stationPtsEl.innerText = `${stationPoints} / 8`;

        // Station Proximity & Boarding
        let nextSt = stations.find(s => s.pos > trainPos - 50) || stations[stations.length - 1];
        let dist = Math.max(0, nextSt.pos - trainPos);

        nextStationEl.innerText = nextSt.name;
        stationDistEl.innerText = `${Math.round(dist)}m`;

        const currentStation = stations.find(s => Math.abs(s.pos - trainPos) < 60);

        if (currentStation && Math.abs(trainSpeed) < 0.2 && doorsOpen) {
            boardingCooldown++;

            // Award 1 Station Point max per station stop
            if (!currentStation.visited && !currentStation.isTerminal) {
                currentStation.visited = true;
                stationPoints++;
                showLog(`⭐ +1 Station Point earned at ${currentStation.name}!`);
            }

            // Passenger Boarding / Offloading Logic
            if (boardingCooldown % 25 === 0) {
                if (currentStation.isTerminal) {
                    // Disembark all at Grand Terminal
                    if (passengersOnBoard > 0) {
                        passengersDelivered += passengersOnBoard;
                        showLog(`🎉 All ${passengersOnBoard} passengers delivered to Grand Terminal!`);
                        passengersOnBoard = 0;
                        playChimeSound();
                        setTimeout(triggerTerminalFinish, 1200);
                    } else {
                        triggerTerminalFinish();
                    }
                } else if (currentStation.passengers > 0) {
                    // Board passengers onto train
                    currentStation.passengers--;
                    passengersOnBoard++;
                    playChimeSound();
                    showLog(`😃 Passenger boarded! (${currentStation.passengers} left at station)`);
                }
            }
        } else {
            boardingCooldown = 0;
        }

        // Render Canvas
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        drawDaytimeBackground(cameraX);
        drawTrackAndScenery(cameraX);
        drawStationsSideView(cameraX);
        drawTrainSideView(cameraX);

        requestAnimationFrame(update);
    }

    requestAnimationFrame(update);
